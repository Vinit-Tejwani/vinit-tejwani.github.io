  // Reveal timeline rows and general sections as they scroll into view

  const timelineRows = document.querySelectorAll('.timeline-row');

  const revealEls = document.querySelectorAll('.reveal');


  const revealObserver = new IntersectionObserver((entries) => {

    entries.forEach(entry => {

      if(entry.isIntersecting){

        entry.target.classList.add('is-active', 'is-visible');

        revealObserver.unobserve(entry.target);

      }

    });

  }, {threshold:0.2, rootMargin:'0px 0px -8% 0px'});


  timelineRows.forEach(row => revealObserver.observe(row));

  revealEls.forEach(el => revealObserver.observe(el));


  // Scroll-spy: highlight the nav link for whichever section is in view

  const sections = ['work','projects','skills','education','contact']

    .map(id => document.getElementById(id))

    .filter(Boolean);


  const spyObserver = new IntersectionObserver((entries) => {

    entries.forEach(entry => {

    function tick(now){

      const progress = Math.min((now - start) / duration, 1);

      const eased = 1 - Math.pow(1 - progress, 3);

      const value = target * eased;
